        async void OnMessage(object sender,CoreWebView2WebMessageReceivedEventArgs e){

                if(action=="window"){var a=args.Length>0?Convert.ToString(args[0]):"";if(a=="minimize")WindowState=FormWindowState.Minimized;if(a=="maximize")WindowState=WindowState==FormWindowState.Maximized?FormWindowState.Normal:FormWindowState.Maximized;if(a=="close")Close();if(a=="start-drag"){ReleaseCapture();SendMessage(Handle,0xA1,new IntPtr(2),IntPtr.Zero);}if(a=="test-resize"&&Environment.GetEnvironmentVariable("NZIP_TEST_MODE")=="1"&&args.Length==3)ClientSize=new Size(Convert.ToInt32(args[1]),Convert.ToInt32(args[2]));result=Json.Obj("ok",true);}

                else if(action=="choose")result=Choose(args);
                else if(action=="copy-text"){Clipboard.SetText(Convert.ToString(args[0]));result=Json.Obj("ok",true);}
                else result=await Task.Run(()=>backend.Dispatch(action,args));
            }catch(Exception error){result=ArchiveError.Failure(error);}Post(Json.Encode(Json.Obj("kind","reply","id",id,"result",result)));
        }
        J Choose(object[] args){var kind=args.Length>0?Convert.ToString(args[0]):"";var defaultPath=args.Length>1?Convert.ToString(args[1]):"";string[] paths=new string[0];
            if(kind=="folder"||kind=="destination"){using(var d=new FolderBrowserDialog{Description="选择文件夹",SelectedPath=defaultPath??"",ShowNewFolderButton=true})if(d.ShowDialog(this)==DialogResult.OK)paths=new[]{d.SelectedPath};}

            return Json.Obj("ok",true,"paths",paths);
        }
        async void OnClosing(object sender,FormClosingEventArgs e){if(backend.Busy){e.Cancel=true;if(exitPending)return;if(MessageBox.Show(this,"还有任务未完成。是否取消任务并退出？临时文件将被清理。","NZIP",MessageBoxButtons.YesNo,MessageBoxIcon.Question)!=DialogResult.Yes)return;exitPending=true;backend.CancelAll();while(backend.Busy)await Task.Delay(30);Close();}}
